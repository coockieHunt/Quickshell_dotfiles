import QtQuick
import "../info"
import "../../theme"

Rectangle {
    id: root
    property string label: ""
    property string iconFont: Theme.iconFont
    property string tooltip: ""
    property int iconSize: Theme.iconSize
    property bool highlighted: false
    property color idleColor: highlighted ? Theme.accent : "transparent"
    property color hoverColor: "transparent"
    property color iconColor: highlighted ? Theme.textOnAccent : Theme.text
    property color hoverIconColor: Theme.accent
    readonly property bool hovered: mouse.containsMouse
    signal clicked()
    signal rightClicked()
    signal middleClicked()

    width: Theme.buttonSize
    height: Theme.buttonSize
    radius: Theme.cornerRadius
    color: mouse.containsMouse ? hoverColor : idleColor

    PopupTooltip {
        anchorItem: root
        text: root.tooltip
        shown: mouse.containsMouse
    }

    Text {
        anchors.centerIn: parent
        text: root.label
        color: mouse.containsMouse ? root.hoverIconColor : root.iconColor
        font.family: root.iconFont
        font.pixelSize: root.iconSize

        Behavior on color {
            ColorAnimation { duration: 120 }
        }
    }

    MouseArea {
        id: mouse
        anchors.fill: parent
        acceptedButtons: Qt.LeftButton | Qt.RightButton | Qt.MiddleButton
        hoverEnabled: true
        cursorShape: Qt.PointingHandCursor
        onClicked: mouse => {
            if (mouse.button === Qt.RightButton)
                root.rightClicked()
            else if (mouse.button === Qt.MiddleButton)
                root.middleClicked()
            else
                root.clicked()
        }
    }
}

import QtQuick
import Quickshell
import "../../theme"

PopupWindow {
    id: root

    required property Item anchorItem
    property string text: ""
    property string title: ""
    property string icon: ""
    property bool shown: false
    property var rows: []

    anchor.window: anchorItem.QsWindow.window
    anchor.adjustment: PopupAdjustment.None
    anchor.gravity: Edges.Bottom | Edges.Right
    anchor.onAnchoring: {
        const content = anchorItem.QsWindow.contentItem
        const pos = content.mapFromItem(
            anchorItem,
            anchorItem.width / 2 - root.width / 2,
            anchorItem.height + 4
        )
        const maxX = Math.max(4, content.width - root.width - 4)
        anchor.rect.x = Math.max(4, Math.min(pos.x, maxX))
        anchor.rect.y = pos.y + Theme.dropdownGap
    }

    visible: shown && (text.length > 0 || rows.length > 0)
    color: "transparent"
    implicitWidth: 264
    implicitHeight: (title.length > 0
        ? tooltipHeader.implicitHeight
            + (rows.length > 0 ? tooltipRows.implicitHeight + 8
                : text.length > 0 ? tooltipText.implicitHeight + 8 : 0)
        : rows.length > 0 ? tooltipRows.implicitHeight : tooltipText.implicitHeight) + 20

    Rectangle {
        anchors.fill: parent
        color: Theme.componentBackground
        radius: Theme.cornerRadius
        border.color: Theme.componentBorder
        border.width: 1

    Column {
        anchors.centerIn: parent

            id: tooltipHeader
            spacing: 7

                text: root.icon
